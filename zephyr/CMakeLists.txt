# The public headers select instance layouts and initializers on
# ARM_MATH_AUTOVECTORIZE and DISABLEFLOAT16, so every source file that includes
# them must see the library's values. They are defined in a generated
# arm_dsp_config.h, which arm_math_types.h includes ahead of its feature tests
# when ARM_DSP_CUSTOM_CONFIG is defined; other libraries in the image do not
# see them. With CONFIG_CMSIS_DSP_CUSTOM_CONFIG the generated header goes on to
# the user's arm_dsp_config.h with #include_next, so its directory is placed
# ahead of the other include directories.
set(CMSIS_DSP_CONFIG_DIR ${CMAKE_CURRENT_BINARY_DIR}/cmsis_dsp_config)
set(CMSIS_DSP_CONFIG_BODY "")
if(CONFIG_CMSIS_DSP_AUTOVECTORIZE)
  string(APPEND CMSIS_DSP_CONFIG_BODY "#ifndef ARM_MATH_AUTOVECTORIZE\n#define ARM_MATH_AUTOVECTORIZE\n#endif\n")
endif()
if(NOT CONFIG_CMSIS_DSP_FLOAT16)
  string(APPEND CMSIS_DSP_CONFIG_BODY "#ifndef DISABLEFLOAT16\n#define DISABLEFLOAT16\n#endif\n")
endif()
if(CONFIG_CMSIS_DSP_CUSTOM_CONFIG)
  string(APPEND CMSIS_DSP_CONFIG_BODY "#include_next \"arm_dsp_config.h\"\n")
endif()
file(CONFIGURE OUTPUT ${CMSIS_DSP_CONFIG_DIR}/arm_dsp_config.h CONTENT
"/* Generated by the CMSIS-DSP Zephyr module from Kconfig. */
#ifndef CMSIS_DSP_ZEPHYR_CONFIG_H_
#define CMSIS_DSP_ZEPHYR_CONFIG_H_
${CMSIS_DSP_CONFIG_BODY}#endif
" @ONLY)
target_include_directories(zephyr_interface BEFORE INTERFACE ${CMSIS_DSP_CONFIG_DIR})
zephyr_compile_definitions(ARM_DSP_CUSTOM_CONFIG)
